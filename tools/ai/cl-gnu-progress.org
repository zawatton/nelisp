#+TITLE: NeLisp cl layer on the vendored GNU 31.1 files (pcase, gv, cl-macs, cl-generic)

# Design and user decisions: tools/ai/cl-gnu-layer-design.md (sections 4 and 6).
# Policy: never reimplement or shim what a vendored GNU file provides; stage the
# GNU sources verbatim.  The native versions stay until the GNU ones pass the
# differential corpus and the timing check, then they are deleted.
# Stage SN corresponds to design step PN.  P0 (ledger, corpus, baselines) is
# folded into S1.  CLGNU_BIN is a binary built from this tree.  Criteria that
# need the binary are SKIP without it.  A criterion that currently fails on the
# latest binary stays a failing cmd.
# Deletion gates require the FULL GNU match of the relevant corpus (user: match GNU; never accept a partial count).
# Corpus: tools/ai/cl-gnu/corpus-*.el, expected from GNU Emacs 31.1 in
# expected-*.txt, compared by tools/ai/cl-gnu/compare.sh.  Baseline timings:
# tools/ai/cl-gnu/baseline-build140.txt, checked by timing-check.sh.

* S1 P0/P1 baselines, differential corpus and verbatim GNU sources

** S1.1 The GNU 31.1 expected files exist for all four corpora with no error rows
cmd: test "$(cat tools/ai/cl-gnu/expected-*.txt | wc -l)" -ge 60 && ! grep -q ERR tools/ai/cl-gnu/expected-*.txt && ls tools/ai/cl-gnu/corpus-pcase.el tools/ai/cl-gnu/corpus-defstruct.el tools/ai/cl-gnu/corpus-generic.el tools/ai/cl-gnu/corpus-setf.el >/dev/null

** S1.2 The expected files are reproducible on host GNU Emacs 31.1
#+TIMEOUT: 120
cmd: d=$(mktemp -d) && for c in pcase defstruct generic setf; do (ulimit -v 8388608; CLGNU_CORPUS=tools/ai/cl-gnu/corpus-$c.el timeout 100 emacs --batch -Q -l tools/ai/cl-gnu/run-corpus.el 2>/dev/null) | grep -E '^[0-9][0-9] ' | cmp - tools/ai/cl-gnu/expected-$c.txt || exit 1; done

** S1.3 Performance baseline for build-on140 is recorded (cold start, pcase loop, bytecomp require)
evidence: tools/ai/cl-gnu/baseline-build140.txt sha256=6c3ccb30d5d6b1d056b97d8f4b92ed470ab50e8ae37dd9b50e87ac4b98bb1fb9

** S1.4 The five GNU sources are vendored byte-for-byte under vendor/staged-emacs-lisp
cmd: for f in pcase cl-lib cl-generic cl-extra cl-print; do zcat /usr/local/share/emacs/31.1/lisp/emacs-lisp/$f.el.gz | cmp - vendor/staged-emacs-lisp/$f.el && grep -q "^staged-emacs-lisp/$f.el	GNU Emacs 31.1" vendor/ORIGIN && grep -q "vendor/staged-emacs-lisp/$f.el\$" vendor/SHA256SUMS || exit 1; done; grep -E 'staged-emacs-lisp/(pcase|cl-lib|cl-generic|cl-extra|cl-print)\.el$' vendor/SHA256SUMS | sha256sum -c --quiet

** S1.5 All four corpora are identical to GNU 31.1 (pcase 31/31, defstruct 38/38, generic 53/53, setf 36/36)
#+REQUIRES: CLGNU_BIN
cmd: bash tools/ai/cl-gnu/compare.sh

** S1.6 End to end: the binary still loads its own build dependencies (selfhost probe-load)
#+REQUIRES: CLGNU_BIN
cmd: SELFHOST_BIN="$CLGNU_BIN" bash tools/ai/selfhost-build/run.sh probe-load

* S2 P2 stage static-if

** S2.1 static-if is staged from vendor/staged-emacs-lisp/subr.el in the build script
cmd: grep -q "vendor/staged-emacs-lisp/subr.el\" 'static-if)" scripts/nelisp-standalone-build.el

** S2.2 The binary provides static-if with GNU semantics
#+REQUIRES: CLGNU_BIN
cmd: "$CLGNU_BIN" --load tools/ai/cl-gnu/probe-static-if.el 2>&1 | grep -aq '^PROBE-STATIC-IF-PASS'

** S2.3 cl-macs.el loads form by form with at most 2 failures (3 before static-if)
#+REQUIRES: CLGNU_BIN
cmd: "$CLGNU_BIN" --load tools/ai/cl-gnu/probe-forms.el -- vendor/staged-emacs-lisp/cl-macs.el 2>&1 | grep -a '^PROBE-FORMS' | grep -aq 'failures=[0-2]$'

** S2.4 The static-if form no longer appears among the cl-macs failures
#+REQUIRES: CLGNU_BIN
cmd: ! "$CLGNU_BIN" --load tools/ai/cl-gnu/probe-forms.el -- vendor/staged-emacs-lisp/cl-macs.el 2>&1 | grep -aq 'void-function (static-if\|(void-function static-if)'

** S2.5 End to end: static-if is available, all corpora match GNU in full and the selfhost probe passes
#+REQUIRES: CLGNU_BIN
cmd: "$CLGNU_BIN" --load tools/ai/cl-gnu/probe-static-if.el 2>&1 | grep -aq '^PROBE-STATIC-IF-PASS' && bash tools/ai/cl-gnu/compare.sh && SELFHOST_BIN="$CLGNU_BIN" bash tools/ai/selfhost-build/run.sh probe-load
